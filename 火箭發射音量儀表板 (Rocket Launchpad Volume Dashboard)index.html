    /* ==========================================================================

       4. Web Audio API 音量偵測與例外處理 (P0 & P2)

       ========================================================================== */

    async function startMic() {

      try {

        errorBanner.style.display = 'none';

        

        // 請求麥克風權限

        const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });

        micStream = stream;


        // 建立 AudioContext 與 AnalyserNode

        audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        const source = audioCtx.createMediaStreamSource(stream);

        analyser = audioCtx.createAnalyser();

        analyser.fftSize = 256;


        source.connect(analyser);

        dataArray = new Uint8Array(analyser.frequencyBinCount);


        isMicActive = true;

        btnMicInit.style.display = 'none';

        statusBadge.textContent = "🟢 系統運作中，請保持安靜...";

        statusBadge.className = "status-badge charging";


        // 啟動音量計算主迴圈

        requestAnimationFrame(processAudio);

      } catch (err) {

        console.error("麥克風啟動失敗：", err);

        if (err.name === 'NotAllowedError' || err.name === 'PermissionDeniedError') {

          errorText.textContent = "麥克風權限被拒絕，請允許存取麥克風後重新整理頁面。";

        } else {

          errorText.textContent = "未檢測到麥克風裝置，請確認已連接收音設備。";

        }

        errorBanner.style.display = 'flex';

      }

    }


    /* ==========================================================================

       5. 核心計算主迴圈 (包含安靜累積與緩慢倒扣邏輯) (P0)

       ========================================================================== */

    let lastTime = performance.now();


    function processAudio(now) {

      const deltaTime = (now - lastTime) / 1000; // 換算成秒數

      lastTime = now;


      if (isMicActive && !isReadyToLaunch && !isLaunching) {

        // A. 計算當前 RMS 音量 (0~100%)

        analyser.getByteFrequencyData(dataArray);

        let sum = 0;

        for (let i = 0; i < dataArray.length; i++) {

          sum += dataArray[i];

        }

        let average = sum / dataArray.length;

        let currentVolume = Math.min(100, Math.round((average / 128) * 100));


        // B. 更新即時音量計

        volumeMeter.style.width = `${currentVolume}%`;

        volumeValue.textContent = `${currentVolume}%`;


        // C. 判斷是否低於安靜門檻

        const isQuiet = currentVolume <= sensitivityThreshold;


        if (isQuiet) {

          // --- 安靜狀態：集氣中 ---

          volumeMeter.classList.remove('exceeded');

          bgOverlay.className = 'bg-overlay charging';

          statusBadge.textContent = "✨ 穩定集氣中...";

          statusBadge.className = "status-badge charging";


          // 火焰動態：越安靜，火焰越高

          let quietnessRatio = (sensitivityThreshold - currentVolume) / sensitivityThreshold; 

          let flameHeight = 20 + quietnessRatio * 80; // 20px ~ 100px

          flameContainer.style.height = `${flameHeight}px`;


          // 燃料累積：(100 / 目標秒數) % / 秒

          let gainRate = 100 / targetTimeSeconds;

          fuelPercent = Math.min(100, fuelPercent + gainRate * deltaTime);


        } else {

          // --- 太吵狀態：燃料流失 ---

          volumeMeter.classList.add('exceeded');

          bgOverlay.className = 'bg-overlay warning';

          statusBadge.textContent = "⚠️ 太吵囉！燃料流失中...";

          statusBadge.className = "status-badge warning";


          // 火焰縮小

          flameContainer.style.height = '10px';


          // 播放警告短音

          playBeepSound(220, 0.05, 'sawtooth');


          // 燃料流失：累積速度的一半 (流失完畢需 2 * targetTimeSeconds 秒)

          let lossRate = 100 / (targetTimeSeconds * 2);

          fuelPercent = Math.max(0, fuelPercent - lossRate * deltaTime);

        }


        // D. 更新燃料 UI

        fuelMeter.style.width = `${fuelPercent}%`;

        fuelValue.textContent = `${Math.floor(fuelPercent)}%`;


        // E. 燃料滿 100% 觸發手動發射模式

        if (fuelPercent >= 100) {

          triggerReadyToLaunch();

        }

      }


      requestAnimationFrame(processAudio);

    }

    // 3, 2, 1 倒數

    function startCountdown() {

      let count = 3;

      countdownOverlay.textContent = count;

      countdownOverlay.classList.add('show');

      playBeepSound(523.25, 0.15, 'sine');


      const timer = setInterval(() => {

        count--;

        if (count > 0) {

          countdownOverlay.textContent = count;

          playBeepSound(523.25, 0.15, 'sine');

        } else if (count === 0) {

          countdownOverlay.textContent = "🚀";

          playBeepSound(1046.5, 0.4, 'triangle');

        } else {

          clearInterval(timer);

          countdownOverlay.classList.remove('show');

          executeLaunch();

        }

      }, 1000);

    }

    // 嗶嗶聲與警示音合成

    function playBeepSound(freq, duration, type = 'sine') {

      if (!audioCtx) return;

      try {

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = type;

        osc.frequency.value = freq;

        

        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);


        osc.connect(gain);

        gain.connect(audioCtx.destination);

        

        osc.start();

        osc.stop(audioCtx.currentTime + duration);

      } catch(e) {}

    }


    // 粉紅噪音 (Pink Noise) + 低通濾波器模擬火箭引擎轟鳴聲

    function playRocketEngineSound(duration) {

      if (!audioCtx) return;

      try {

        const bufferSize = audioCtx.sampleRate * duration;

        const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

        const output = buffer.getChannelData(0);

        

        // 粉紅噪音生成演算法

        let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;

        for (let i = 0; i < bufferSize; i++) {

          let white = Math.random() * 2 - 1;

          b0 = 0.99886 * b0 + white * 0.0555179;

          b1 = 0.99332 * b1 + white * 0.0750759;

          b2 = 0.96900 * b2 + white * 0.1538520;

          b3 = 0.86650 * b3 + white * 0.3104856;

          b4 = 0.55000 * b4 + white * 0.5329522;

          b5 = -0.7616 * b5 - white * 0.0168980;

          output[i] = b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362;

          output[i] *= 0.11;

          b6 = white * 0.115926;

        }


        const whiteNoise = audioCtx.createBufferSource();

        whiteNoise.buffer = buffer;


        // 動態低通濾波器 ( Low-pass Filter )

        const filter = audioCtx.createBiquadFilter();

        filter.type = 'lowpass';

        filter.frequency.setValueAtTime(300, audioCtx.currentTime);

        filter.frequency.linearRampToValueAtTime(800, audioCtx.currentTime + 1.0);

        filter.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + duration);


        const gain = audioCtx.createGain();

        gain.gain.setValueAtTime(0.5, audioCtx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);


        whiteNoise.connect(filter);

        filter.connect(gain);

        gain.connect(audioCtx.destination);


        whiteNoise.start();

      } catch(e) {}

    }
